title: 'List Estimates'
description: 'Get a paginated list of estimates'

Get a paginated list of estimates

```http
GET https://services.leadconnectorhq.com/invoices/estimate/list
```

## Autorización

<ParamField header="Authorization" type="string" required>
Bearer token generado desde el portal Leadway. Ver [Autenticación](/authentication).
</ParamField>

<ParamField header="Version" type="string" required default="2021-07-28">
Versión de la API.
</ParamField>

## Query parameters

<ParamField query="altId" type="string" required default='6578278e879ad2646715ba9c'>
Account Id or Agency Id
</ParamField>

<ParamField query="altType" type="string" required>
</ParamField>

<ParamField query="startAt" type="string" default='2023-01-01'>
startAt in YYYY-MM-DD format
</ParamField>

<ParamField query="endAt" type="string" default='2023-01-01'>
endAt in YYYY-MM-DD format
</ParamField>

<ParamField query="search" type="string" default='Home services estimate'>
search text for estimates name
</ParamField>

<ParamField query="status" type="string" default='sent'>
estimate status
</ParamField>

<ParamField query="contactId" type="string" default='AmuzcoPBpgKeccNsFlib'>
Contact ID for the estimate
</ParamField>

<ParamField query="limit" type="string" required default='10'>
Limit the number of items to return
</ParamField>

<ParamField query="offset" type="string" required default='10'>
Number of items to skip
</ParamField>
