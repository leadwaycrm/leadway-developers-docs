Post the necessary fields for the API to add a new inbound message. &lt;br /&gt;

```http
POST https://services.leadconnectorhq.com/conversations/messages/inbound
```

## Autorización

<ParamField header="Authorization" type="string" required>
Bearer token generado desde el portal Leadway. Ver [Autenticación](/authentication).
</ParamField>

<ParamField header="Version" type="string" required default="2021-07-28">
Versión de la API.
</ParamField>

## Body

<ParamField body="type" type="string" required default='SMS'>
Message Type Posibles valores: 'SMS', 'RCS', 'Email', 'WhatsApp', 'GMB', 'IG', 'FB', 'Custom', 'WebChat', 'Live_Chat', 'Call', 'IVR_Call', 'Campaign_Call', 'Campaign_VoiceMail', 'TIKTOK', 'ALL_IN_ONE_CHAT', 'FORM_SUBMISSION'
</ParamField>

<ParamField body="attachments" type="string[]">
Array of attachments
</ParamField>

<ParamField body="message" type="string">
Message Body
</ParamField>

<ParamField body="conversationId" type="string" required default='ve9EPM428h8vShlRW1KT'>
Conversation Id
</ParamField>

<ParamField body="contactId" type="string" required default='ve9EPM428h8vShlRW1KT'>
Contact Id
</ParamField>

<ParamField body="conversationProviderId" type="string" required default='61d6d1f9cdac7612faf80753'>
Conversation Provider Id
</ParamField>

<ParamField body="html" type="string">
HTML Body of Email
</ParamField>

<ParamField body="subject" type="string">
Subject of the Email
</ParamField>

<ParamField body="emailFrom" type="string" default='sender@company.com'>
Email address to send from. This field is associated with the contact record and cannot be dynamically changed.
</ParamField>

<ParamField body="emailTo" type="string">
Recipient email address. This field is associated with the contact record and cannot be dynamically changed.
</ParamField>

<ParamField body="emailCc" type="string[]">
List of email address to CC
</ParamField>

<ParamField body="emailBcc" type="string[]">
List of email address to BCC
</ParamField>

<ParamField body="emailMessageId" type="string">
Send the email message id for which this email should be threaded. This is for replying to a specific email
</ParamField>

<ParamField body="altId" type="string" default='61d6d1f9cdac7612faf80753'>
external mail provider's message id
</ParamField>

<ParamField body="direction" type="object">
Message direction, if required can be set manually, default is outbound
</ParamField>

<ParamField body="date" type="string">
Date of the inbound message
</ParamField>

<ParamField body="call" type="object">
Phone call dialer and receiver information
</ParamField>
